package burgershop;

public class Burger {

    private String orderId;
    private String customerId;
    private String customerName;
    private int qty;
    private int status;

    public Burger(String orderId, String customerId, String customerName, int qty, int status) {
        this.orderId = orderId;
        this.customerId = customerId;
        this.customerName = customerName;
        this.qty = qty;
        this.status = status;
    }

    public String getOrderId() {
        return orderId;
    }

    public String getCustomerId() {
        return customerId;
    }

    public String getCustomerName() {
        return customerName;
    }

    public int getQty() {
        return qty;
    }

    public int getStatus() {
        return status;
    }

    public String getStatusString() {
        return getStatusString(status);
    }

    public static String getStatusString(int status) {
        if (status == 0) {
            return "Preparing";
        } else if (status == 1) {
            return "Delivered";
        } else {
            return "Canceled";
        }
    }

    @Override
    public String toString() {
        return orderId + "-" + customerId + "-" + customerName + "-" + qty + "-" + status;
    }

    /**
     * @return the orderStatus
     */
    public int getOrderStatus() {
        return status;
    }

    /**
     * @param orderStatus the orderStatus to set
     */
    public void setOrderStatus(int orderStatus) {
        this.status = orderStatus;
    }

    public boolean equals(Burger burger) {
        return burger != null && orderId != null && orderId.equalsIgnoreCase(burger.orderId);
    }

    @Override
    public boolean equals(Object obj) {
        if (this == obj) {
            return true;
        }
        if (!(obj instanceof Burger)) {
            return false;
        }
        return equals((Burger) obj);
    }

    public void setOrderId(String orderId) {
        this.orderId = orderId;
    }

    public void setCustomerId(String customerId) {
        this.customerId = customerId;
    }

    public void setCustomerName(String customerName) {
        this.customerName = customerName;
    }

    public void setQty(int qty) {
        this.qty = qty;
    }

    public void setStatus(int status) {
        this.status = status;
    }

    public int compareByQty(Burger other) {
        return Integer.compare(this.qty, other.qty);
    }

    public boolean isSameCustomer(Burger other) {
        return this.customerId.equalsIgnoreCase(other.customerId);
    }
}

package burgershop;

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.FileReader;
import java.io.FileWriter;
import java.io.IOException;

public class BurgerController {

    private static final String DATA_FILE = "Burger.txt";

    public static String generateOrderId() {
        int maxOrderIdNum = 0;
        try (BufferedReader br = new BufferedReader(new FileReader(DATA_FILE))) {
            String line;
            while ((line = br.readLine()) != null) {
                String trimmed = line.trim();
                if (trimmed.isEmpty()) {
                    continue;
                }
                String[] parts = trimmed.split(",");
                String id = parts[0].trim();
                if (id.length() >= 2 && (id.charAt(0) == 'B' || id.charAt(0) == 'b')) {
                    try {
                        int num = Integer.parseInt(id.substring(1));
                        if (num > maxOrderIdNum) {
                            maxOrderIdNum = num;
                        }
                    } catch (NumberFormatException ignored) {
                    }
                }
            }
        } catch (IOException ex) {
            // File may not exist yet before the first order is placed
        }

        return String.format("B%04d", maxOrderIdNum + 1);
    }

    public static boolean isCustomerIdValid(String customerId) {
        if (customerId == null || customerId.length() != 10) {
            return false;
        }
        if (customerId.charAt(0) != '0') {
            return false;
        }
        for (int i = 0; i < customerId.length(); i++) {
            if (!Character.isDigit(customerId.charAt(i))) {
                return false;
            }
        }
        return true;
    }

    public static boolean isOrderPlaced(String orderId, String customerId, String customerName, String qty) {
        if (orderId == null || customerId == null || customerName == null || qty == null) {
            return false;
        }
        String cleanName = customerName.trim();
        if (cleanName.isEmpty() || cleanName.contains(",")) {
            return false;
        }
        int parsedQty;
        try {
            parsedQty = Integer.parseInt(qty.trim());
            if (parsedQty < 1) {
                return false;
            }
        } catch (NumberFormatException e) {
            return false;
        }

        try (BufferedWriter bw = new BufferedWriter(new FileWriter(DATA_FILE, true))) {
            bw.write(orderId.trim() + "," + customerId.trim() + "," + cleanName + "," + parsedQty + "," + 0);
            bw.newLine();
            return true;
        } catch (IOException e) {
            return false;
        }
    }

    public static List allBurgerOrdersList() {
        List allBurgerList = new List();
        try (BufferedReader br = new BufferedReader(new FileReader(DATA_FILE))) {
            String line;
            while ((line = br.readLine()) != null) {
                if (line.trim().isEmpty()) {
                    continue;
                }
                String[] parts = line.split(",");
                if (parts.length == 5) {
                    try {
                        String orderId = parts[0].trim();
                        String customerId = parts[1].trim();
                        String customerName = parts[2].trim();
                        int qty = Integer.parseInt(parts[3].trim());
                        int status = Integer.parseInt(parts[4].trim());

                        Burger burger = new Burger(orderId, customerId, customerName, qty, status);
                        allBurgerList.addLast(burger);
                    } catch (NumberFormatException ignored) {
                    }
                }
            }
        } catch (IOException ex) {
            // File may not exist yet before the first order is placed
        }
        return allBurgerList;
    }

    // 0=Preparing 1=Delivered 2=Canceled
    public static List allBurgerOrdersListByStatus(int status) {
        List allBurgerOrdersList = allBurgerOrdersList();
        List filteredBurgerList = new List();

        for (int i = 0; i < allBurgerOrdersList.size(); i++) {
            Burger b = allBurgerOrdersList.get(i);
            if (b.getStatus() == status) {
                filteredBurgerList.addLast(b);
            }
        }

        return filteredBurgerList;
    }

    public static List getAllOrdersTopToLow() {
        List allOrders = allBurgerOrdersList(); // main data
        List tempList = new List();
        List uniqueCustomerList = new List(); // holds each unique customer once
        List sortedOrders = new List();

        // Step 1: copy allOrders into tempList
        for (int i = 0; i < allOrders.size(); i++) {
            tempList.addLast(allOrders.get(i));
        }

        // Step 2: build uniqueCustomerList (only customerIds, qty = 0)
        for (int i = 0; i < tempList.size(); i++) {
            Burger b = tempList.get(i);
            boolean exists = false;

            for (int j = 0; j < uniqueCustomerList.size(); j++) {
                Burger existing = uniqueCustomerList.get(j);
                if (existing.getCustomerId().equalsIgnoreCase(b.getCustomerId())) {
                    exists = true;
                    break;
                }
            }

            if (!exists) {
                uniqueCustomerList.addLast(new Burger("", b.getCustomerId(), b.getCustomerName(), 0, 0));
            }
        }

        // Step 3: sum qty per unique customer
        for (int i = 0; i < uniqueCustomerList.size(); i++) {
            Burger unique = uniqueCustomerList.get(i);
            int totalQty = 0;

            for (int j = 0; j < tempList.size(); j++) {
                Burger order = tempList.get(j);
                if (order.getCustomerId().equalsIgnoreCase(unique.getCustomerId())) {
                    totalQty += order.getQty();
                }
            }

            unique.setQty(totalQty); // store total in qty field
        }

        // Step 4: sort uniqueCustomerList manually (descending qty)
        for (int i = 0; i < uniqueCustomerList.size() - 1; i++) {
            for (int j = i + 1; j < uniqueCustomerList.size(); j++) {
                Burger a = uniqueCustomerList.get(i);
                Burger b = uniqueCustomerList.get(j);

                if (b.getQty() > a.getQty()) {
                    // swap
                    Burger temp = a;
                    uniqueCustomerList.set(i, b);
                    uniqueCustomerList.set(j, temp);
                }
            }
        }

        // Step 5: build sortedOrders by order of sorted customers
        for (int i = 0; i < uniqueCustomerList.size(); i++) {
            Burger topCustomer = uniqueCustomerList.get(i);
            for (int j = 0; j < tempList.size(); j++) {
                Burger b = tempList.get(j);
                if (b.getCustomerId().equalsIgnoreCase(topCustomer.getCustomerId())) {
                    sortedOrders.addLast(b);
                }
            }
        }

        return sortedOrders;
    }

    public static Burger getBurgerByOrderId(String orderId) {
        if (orderId == null || orderId.trim().isEmpty()) {
            return null;
        }
        String target = orderId.trim();
        List allOrders = allBurgerOrdersList();
        for (int i = 0; i < allOrders.size(); i++) {
            Burger b = allOrders.get(i);
            if (b.getOrderId().equalsIgnoreCase(target)) {
                return b;
            }
        }
        return null;
    }

    public static List getOrdersByCustomerId(String customerId) {
        List matchingOrders = new List();
        if (customerId == null || customerId.trim().isEmpty()) {
            return matchingOrders;
        }
        String target = customerId.trim();
        List allOrders = allBurgerOrdersList();
        for (int i = 0; i < allOrders.size(); i++) {
            Burger b = allOrders.get(i);
            if (b.getCustomerId().equalsIgnoreCase(target)) {
                matchingOrders.addLast(b);
            }
        }
        return matchingOrders;
    }

    public static boolean updateOrderDetails(String orderId, int newQty, int newStatus) {
        if (orderId == null || orderId.trim().isEmpty() || newQty < 1 || newStatus < 0 || newStatus > 2) {
            return false;
        }
        String target = orderId.trim();
        List allOrders = allBurgerOrdersList();
        boolean updated = false;

        for (int i = 0; i < allOrders.size(); i++) {
            Burger b = allOrders.get(i);
            if (b.getOrderId().equalsIgnoreCase(target)) {
                if (b.getStatus() != 0) {
                    return false;
                }
                b.setQty(newQty);
                b.setStatus(newStatus);
                updated = true;
                break;
            }
        }

        if (!updated) {
            return false;
        }

        try (BufferedWriter bw = new BufferedWriter(new FileWriter(DATA_FILE, false))) {
            for (int i = 0; i < allOrders.size(); i++) {
                Burger b = allOrders.get(i);
                bw.write(b.getOrderId() + "," + b.getCustomerId() + "," + b.getCustomerName() + "," + b.getQty() + "," + b.getStatus());
                bw.newLine();
            }
            return true;
        } catch (IOException ex) {
            return false;
        }
    }

}

    private void btnSearchActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnSearchActionPerformed
        String orderId = txtOrderId.getText().trim();
        if (orderId.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please enter an Order ID.", "Info", JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        Burger b = BurgerController.getBurgerByOrderId(orderId);

        if (b == null) {
            editingOrderId = "";
            txtCustomerId.setText("");
            txtCustomerName.setText("");
            txtBurgerQty.setText("");
            txtBurgerQty.setEditable(false);
            comboBox.setSelectedIndex(0);
            comboBox.setEnabled(false);
            btnSave.setVisible(false);
            JOptionPane.showMessageDialog(this, orderId + " order not found.");
        } else {
            editingOrderId = b.getOrderId();
            txtCustomerId.setText(b.getCustomerId());
            txtCustomerName.setText(b.getCustomerName());
            txtBurgerQty.setText(String.valueOf(b.getQty()));
            comboBox.setSelectedIndex(b.getStatus());
            int orderStatus = b.getStatus();

            if (orderStatus == 0) {
                txtBurgerQty.setEditable(true);
                comboBox.setEnabled(true);
                comboBox.setFocusable(true);
                btnSave.setVisible(true);
            } else {
                txtBurgerQty.setEditable(false);
                comboBox.setEnabled(false);
                btnSave.setVisible(false);
                JOptionPane.showMessageDialog(this, b.getOrderId() + " cannot be updated.\nOnly Preparing orders can be updated.");
            }
        }
    }//GEN-LAST:event_btnSearchActionPerformed

    private void btnSaveActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnSaveActionPerformed
        if (editingOrderId.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please search for an order first.", "Error", JOptionPane.ERROR_MESSAGE);
            return;
        }

        int newQty;
        try {
            newQty = Integer.parseInt(txtBurgerQty.getText().trim());
            if (newQty < 1) {
                JOptionPane.showMessageDialog(this, "Quantity must be greater than 0.", "Error", JOptionPane.ERROR_MESSAGE);
                txtBurgerQty.requestFocusInWindow();
                return;
            }
        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(this, "Please enter a valid number for quantity.", "Error", JOptionPane.ERROR_MESSAGE);
            txtBurgerQty.requestFocusInWindow();
            return;
        }

        int orderStatus = comboBox.getSelectedIndex();
        if (BurgerController.updateOrderDetails(editingOrderId, newQty, orderStatus)) {
            JOptionPane.showMessageDialog(this, editingOrderId + " has been updated.\nOrder Qty : " + newQty + "\nOrder Status : " + Burger.getStatusString(orderStatus));
            if (orderStatus != 0) {
                txtBurgerQty.setEditable(false);
                comboBox.setEnabled(false);
                btnSave.setVisible(false);
                editingOrderId = "";
            }
        } else {
            JOptionPane.showMessageDialog(this, "Failed to update order " + editingOrderId + ".", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }//GEN-LAST:event_btnSaveActionPerformed

    private void addDataToTable() {
        DefaultTableModel model = (DefaultTableModel) jTable2.getModel();
    model.setRowCount(0);

    if (burgerList == null || burgerList.isEmpty()) {
        JOptionPane.showMessageDialog(this, "No orders found.", "Info", JOptionPane.INFORMATION_MESSAGE);
        return;
    }

    // We'll track totals per customer while displaying
    List shownCustomerList = new List();

    for (int i = 0; i < burgerList.size(); i++) {
        Burger b = burgerList.get(i);
        boolean exists = false;

        for (int j = 0; j < shownCustomerList.size(); j++) {
            if (shownCustomerList.get(j).getCustomerId().equalsIgnoreCase(b.getCustomerId())) {
                exists = true;
                break;
            }
        }

        if (!exists) {
            // Calculate total qty for this customer
            int totalQty = 0;
            for (int k = 0; k < burgerList.size(); k++) {
                Burger order = burgerList.get(k);
                if (order.getCustomerId().equalsIgnoreCase(b.getCustomerId())) {
                    totalQty += order.getQty();
                }
            }

            double totalIncome = totalQty * BurgerShop.getBurgerPrice() * 1.0;

            model.addRow(new Object[]{
                model.getRowCount() + 1,
                b.getCustomerId(),
                b.getCustomerName(),
                totalQty,
                totalIncome
            });

            shownCustomerList.addLast(b);
        }
    }

    }

package burgershop;

public class List {
    private Node first;
    
    public boolean add(int index, Burger burger){
        if(index>=0 && index<=size()){
            Node node=new Node(burger);
            if(index==0){
                node.next=first;
                first=node;
            }else{
                int count=0;
                Node temp=first;
                while(count<index-1){
                    temp=temp.next;
                    count++;
                }
                node.next=temp.next;
                temp.next=node;
            }
            return true;
        }
        return false;
    }
    public boolean addFirst(Burger burger){
        return add(0, burger);
    }
    public boolean addLast(Burger burger){
        return add(size(), burger);
    }
    public boolean add(Burger burger){
        return addLast(burger);
    }
    public boolean remove(int index){
        if(index>=0 && index<size()){
            if(index==0){
                first=first.next;
                return true;
            }else{
                int count=0;
                Node temp=first;
                while(count<index-1){
                    count++;
                    temp=temp.next;
                }
                temp.next=temp.next.next;
                return true;
            }
        }
        return false;
    }
    public boolean removeFirst(){
        return remove(0);
    }
    public boolean removeLast(){
        return remove(size()-1);
    }
    public int size(){
        int count=0;
        Node temp=first;
        while(temp!=null){
            temp=temp.next;
            count++;
        }
        return count;
    }
    public void printBurgers(){
        System.out.print("{");
        Node temp=first;
        while(temp!=null){
            Burger burger=temp.burger;
            System.out.print(burger.toString()+", ");
            temp=temp.next;
        }
        System.out.println(isEmpty() ? "empty}":"\b\b}");
    }
    public boolean isEmpty(){
        return first==null;                
    }
    public Burger get(int index){
        if(index>=0 && index<size()){
            int count=0;
            Node temp=first;
            while(count<index){
                count++;
                temp=temp.next;
            }
            return temp.burger;
        }
        return null;
    }
    public Burger getFirst(){
        return get(0);
    }
    public Burger getLast(){
        return get(size()-1);
    }
    public int indexOf(Burger burger){
        if(burger==null){
            return -1;
        }
        int index=0;
        Node temp=first;
        while(temp!=null){
            if(burger.equals(temp.burger)){
                return index;
            }
            index++;
            temp=temp.next;
        }
        return -1;
    }
    public boolean contains(Burger burger){
        return indexOf(burger)!=-1;
    }
    public boolean remove(Burger burger){
        int index=indexOf(burger);
        return remove(index);
    }
    
    public void set(int index, Burger burger) {
    if (index >= 0 && index < size()) {
        int count = 0;
        Node temp = first;
        while (count < index) {
            temp = temp.next;
            count++;
        }
        temp.burger = burger;
    }
}


    
    class Node{
        private Burger burger;
        private Node next;
        public Node(Burger burger){
            this.burger=burger;
        }
    }    
}

    private void btnPlaceOrderActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnPlaceOrderActionPerformed
        String orderId = BurgerController.generateOrderId();
        String customerId = txtCustomerId.getText().trim();
        String customerName = txtCustomerName.getText().trim();
        String qtyText = txtBurgerQty.getText().trim();

        if (!BurgerController.isCustomerIdValid(customerId)) {
            JOptionPane.showMessageDialog(this, "Invalid Customer ID. Please enter a 10-digit number starting with 0.", "Error", JOptionPane.ERROR_MESSAGE);
            txtCustomerId.requestFocusInWindow();
            return;
        }
        if (customerName.isEmpty() || customerName.contains(",")) {
            JOptionPane.showMessageDialog(this, "Invalid Customer Name! Name cannot be empty or contain commas.", "Error", JOptionPane.ERROR_MESSAGE);
            txtCustomerName.requestFocusInWindow();
            return;
        }

        int qty;
        try {
            qty = Integer.parseInt(qtyText);
            if (qty < 1) {
                JOptionPane.showMessageDialog(this, "Quantity must be greater than 0.", "Error", JOptionPane.ERROR_MESSAGE);
                txtBurgerQty.requestFocusInWindow();
                return;
            }
        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(this, "Please enter a valid number for quantity.", "Error", JOptionPane.ERROR_MESSAGE);
            txtBurgerQty.requestFocusInWindow();
            return;
        }

        if (BurgerController.isOrderPlaced(orderId, customerId, customerName, String.valueOf(qty))) {
            JOptionPane.showMessageDialog(this, "Order " + orderId + " placed successfully!", "Success", JOptionPane.INFORMATION_MESSAGE);
            txtCustomerId.setText("");
            txtCustomerName.setText("");
            txtBurgerQty.setText("");
            txtTotal.setText("");
            txtOrderId.setText(BurgerController.generateOrderId());
            txtCustomerId.requestFocusInWindow();
        } else {
            JOptionPane.showMessageDialog(this, "Error while placing Order.", "Error", JOptionPane.ERROR_MESSAGE);
        }
    }//GEN-LAST:event_btnPlaceOrderActionPerformed

    private void btnCalculateActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnCalculateActionPerformed
        String qtyText = txtBurgerQty.getText().trim();
        if (qtyText.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please enter a quantity.", "Error", JOptionPane.ERROR_MESSAGE);
            txtBurgerQty.requestFocusInWindow();
            return;
        }
        try {
            int qty = Integer.parseInt(qtyText);
            if (qty < 1) {
                JOptionPane.showMessageDialog(this, "Quantity must be at least 1.", "Error", JOptionPane.ERROR_MESSAGE);
                txtBurgerQty.requestFocusInWindow();
            } else {
                double price = qty * BurgerShop.getBurgerPrice() * 1.0;
                txtTotal.setText(String.valueOf(price));
            }
        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(this, "Please enter a valid number for quantity.", "Error", JOptionPane.ERROR_MESSAGE);
            txtBurgerQty.requestFocusInWindow();
        }
    }//GEN-LAST:event_btnCalculateActionPerformed

    private void addDataToTable() {
        DefaultTableModel model = (DefaultTableModel) jTable2.getModel();
        model.setRowCount(0);

        for (int i = 0; i < burgerList.size(); i++) {
            Burger b = burgerList.get(i);
            double price = b.getQty() * BurgerShop.getBurgerPrice() * 1.0;
            model.addRow(new Object[]{
                b.getOrderId(),
                b.getCustomerId(),
                b.getQty(),
                price
            });
        }
    }

    private void addDataToTable(List burgers) {
        DefaultTableModel model = (DefaultTableModel) jTable1.getModel();
        model.setRowCount(0);

        for (int i = 0; i < burgers.size(); i++) {
            Burger b = burgers.get(i);
            double price = b.getQty() * BurgerShop.getBurgerPrice() * 1.0;
            model.addRow(new Object[]{
                b.getOrderId(),
                b.getQty(),
                price,
                b.getStatusString()
            });
        }

    }

    private void btnSearchActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnSearchActionPerformed
        String searchedCustomerId = txtCustomerID.getText().trim();
        if (searchedCustomerId.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please enter a Customer ID.", "Info", JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        List matchingOrders = BurgerController.getOrdersByCustomerId(searchedCustomerId);
        addDataToTable(matchingOrders);

        if (matchingOrders.isEmpty()) {
            JOptionPane.showMessageDialog(this, "No orders found.", "Info", JOptionPane.INFORMATION_MESSAGE);
        }
    }//GEN-LAST:event_btnSearchActionPerformed

    private void btnSearchActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnSearchActionPerformed
        String orderId = txtOrderId.getText().trim();
        if (orderId.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please enter an Order ID.", "Info", JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        Burger b = BurgerController.getBurgerByOrderId(orderId);

        if (b == null) {
            JOptionPane.showMessageDialog(this, orderId + " order not found.");
            txtCustomerId.setText("");
            txtCustomerName.setText("");
            txtBurgerQty.setText("");
            txtOrderStatus.setText("");
            txtTotal.setText("");
        } else {
            txtCustomerId.setText(b.getCustomerId());
            txtCustomerName.setText(b.getCustomerName());
            txtBurgerQty.setText(String.valueOf(b.getQty()));
            txtOrderStatus.setText(b.getStatusString());
            txtTotal.setText(String.valueOf(b.getQty() * BurgerShop.getBurgerPrice() * 1.0));
        }

    }//GEN-LAST:event_btnSearchActionPerformed

    private void addDataToTable() {
        DefaultTableModel model = (DefaultTableModel) jTable2.getModel();
        model.setRowCount(0);

        for (int i = 0; i < burgerList.size(); i++) {
            Burger b = burgerList.get(i);
            double price = b.getQty() * BurgerShop.getBurgerPrice() * 1.0;
            model.addRow(new Object[]{
                b.getOrderId(),
                b.getCustomerId(),
                b.getQty(),
                price,
                b.getStatusString()
            });
        }
    }

    private void scaleBurgerImage() {

        ImageIcon originalIcon = new ImageIcon(getClass().getResource("/burgershop/1.png"));

        Image scaledImage = originalIcon.getImage().getScaledInstance(
                imgBurger.getWidth(),
                imgBurger.getHeight(),
                Image.SCALE_SMOOTH
        );

        imgBurger.setIcon(new ImageIcon(scaledImage));
        imgBurger.setText("");
    }

    private void btnCanceledOrdersActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnCanceledOrdersActionPerformed
        List canceledOrders = BurgerController.allBurgerOrdersListByStatus(2);

        if (canceledOrders.size() == 0) {
            JOptionPane.showMessageDialog(null, "No canceled orders found.","Info", JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        new CanceledOrders(canceledOrders).setVisible(true);
    }//GEN-LAST:event_btnCanceledOrdersActionPerformed

    private void btnDeliverdedOrdersActionPerformed(java.awt.event.ActionEvent evt) {//GEN-FIRST:event_btnDeliverdedOrdersActionPerformed
        List deliveredOrders = BurgerController.allBurgerOrdersListByStatus(1);

        if (deliveredOrders.size() == 0) {
            JOptionPane.showMessageDialog(null, "No delivered orders found.", "Info", JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        new DeliverdedOrders(deliveredOrders).setVisible(true);
    }//GEN-LAST:event_btnDeliverdedOrdersActionPerformed
